{
    static int RSA_size(void* rsa) => B.RSA_size((B.rsa_st*)rsa);
    static void RSA_free(void* rsa) => B.RSA_free((B.rsa_st*)rsa);
    static void* EVP_PKEY_get1_RSA(void* key) => B.EVP_PKEY_get1_RSA((B.evp_pkey_st*)key);

        => B.RSA_public_encrypt(length,input,output,(B.rsa_st*)rsa,padding);
    static int Errors() { bool value=false; while(O.ERR_get_error()!=0)value=true; return value?1:0; }

    }
    public static void Run(string fixture)

        modulus.CopyTo(input,0);Raw(rsa,"n",input,size,3);Array.Fill(input,(byte)255);Raw(rsa,"ff",input,size,3);Array.Clear(input);
