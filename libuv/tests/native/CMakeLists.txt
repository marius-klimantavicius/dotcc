cmake_minimum_required(VERSION 3.16)
project(DotccLibuvNative C)
if(NOT LIBUV_SOURCE)
  message(FATAL_ERROR "LIBUV_SOURCE must point at the pinned acquired source")
endif()
set(LIBUV_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LIBUV_BUILD_BENCH OFF CACHE BOOL "" FORCE)
set(LIBUV_BUILD_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBUV_SOURCE}" libuv)
add_executable(libuv-native-workflow workflow.c)
target_link_libraries(libuv-native-workflow PRIVATE uv_a)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(libuv-native-layout ../abi/linux-layout.c)
  target_include_directories(libuv-native-layout PRIVATE "${LIBUV_SOURCE}/include")
endif()
# The native assertions are acceptance checks even in Release builds.
if(MSVC)
  target_compile_options(libuv-native-workflow PRIVATE /UNDEBUG)
else()
  target_compile_options(libuv-native-workflow PRIVATE -UNDEBUG)
endif()
